-- ============================================
--   SmartPark Database Schema (MySQL)

CREATE DATABASE smartpark_db;
USE smartpark_db;

-- Parking Floors / Zones
CREATE TABLE floors (
  floor_id    INT PRIMARY KEY AUTO_INCREMENT,
  floor_code  CHAR(1) NOT NULL UNIQUE,
  floor_name  VARCHAR(50) NOT NULL,
  total_slots INT DEFAULT 0,
  created_at  TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Parking Slots
CREATE TABLE parking_slots (
  slot_id     INT PRIMARY KEY AUTO_INCREMENT,
  slot_code   VARCHAR(10) NOT NULL UNIQUE,
  floor_id    INT,
  slot_type   ENUM('Car', 'Bike', 'EV', 'Handicap') DEFAULT 'Car',
  status      ENUM('available', 'occupied', 'reserved', 'maintenance') 
              DEFAULT 'available',
  hourly_rate DECIMAL(8,2) DEFAULT 30.00,
  FOREIGN KEY (floor_id) REFERENCES floors(floor_id) ON DELETE CASCADE
);

-- Users / Drivers
CREATE TABLE users (
  user_id     INT PRIMARY KEY AUTO_INCREMENT,
  full_name   VARCHAR(100) NOT NULL,
  email       VARCHAR(150) UNIQUE NOT NULL,
  phone       VARCHAR(15),
  password_hash VARCHAR(255),
  role        ENUM('admin', 'driver') DEFAULT 'driver',
  created_at  TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Vehicles
CREATE TABLE vehicles (
  vehicle_id  INT PRIMARY KEY AUTO_INCREMENT,
  user_id     INT,
  plate_no    VARCHAR(20) NOT NULL UNIQUE,
  vehicle_type ENUM('Car', 'Bike', 'EV') DEFAULT 'Car',
  model       VARCHAR(80),
  FOREIGN KEY (user_id) REFERENCES users(user_id)
);

-- Bookings
CREATE TABLE bookings (
  booking_id  INT PRIMARY KEY AUTO_INCREMENT,
  slot_id     INT NOT NULL,
  user_id     INT NOT NULL,
  vehicle_id  INT NOT NULL,
  check_in    DATETIME NOT NULL DEFAULT NOW(),
  check_out   DATETIME,
  duration_hrs DECIMAL(5,2),
  amount      DECIMAL(10,2),
  status      ENUM('active', 'completed', 'cancelled') DEFAULT 'active',
  created_at  TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (slot_id)    REFERENCES parking_slots(slot_id),
  FOREIGN KEY (user_id)    REFERENCES users(user_id),
  FOREIGN KEY (vehicle_id) REFERENCES vehicles(vehicle_id)
);

-- Payments
CREATE TABLE payments (
  payment_id  INT PRIMARY KEY AUTO_INCREMENT,
  booking_id  INT UNIQUE NOT NULL,
  amount      DECIMAL(10,2),
  method      ENUM('Cash','UPI','Card','Wallet') DEFAULT 'UPI',
  paid_at     TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  status      ENUM('pending','success','failed') DEFAULT 'pending',
  FOREIGN KEY (booking_id) REFERENCES bookings(booking_id)
);

-- 1. All available slots with floor info
SELECT
  ps.slot_code,
  f.floor_name,
  ps.slot_type,
  ps.hourly_rate
FROM parking_slots ps
JOIN floors f ON ps.floor_id = f.floor_id
WHERE ps.status = 'available'
ORDER BY f.floor_code, ps.slot_code;

-- 2. Active bookings with user & vehicle details
SELECT
  b.booking_id,
  ps.slot_code,
  u.full_name,
  v.plate_no,
  b.check_in,
  TIMESTAMPDIFF(MINUTE, b.check_in, NOW()) AS mins_parked,
  b.status
FROM bookings b
JOIN parking_slots ps ON b.slot_id    = ps.slot_id
JOIN users         u  ON b.user_id   = u.user_id
JOIN vehicles      v  ON b.vehicle_id = v.vehicle_id
WHERE b.status = 'active';

-- 3. Slot occupancy stats by floor
SELECT
  f.floor_name,
  COUNT(*) AS total,
  SUM(CASE WHEN ps.status = 'available'   THEN 1 ELSE 0 END) AS available,
  SUM(CASE WHEN ps.status = 'occupied'    THEN 1 ELSE 0 END) AS occupied,
  ROUND(SUM(CASE WHEN ps.status='occupied' THEN 1 ELSE 0 END) * 100.0
        / COUNT(*), 1) AS occupancy_pct
FROM parking_slots ps
JOIN floors f ON ps.floor_id = f.floor_id
GROUP BY f.floor_id;

-- 4. Today's revenue
SELECT
  COUNT(*) AS bookings_today,
  SUM(b.amount) AS total_revenue,
  AVG(b.duration_hrs) AS avg_duration_hrs
FROM bookings b
WHERE DATE(b.created_at) = CURDATE()
  AND b.status = 'completed';

-- 5. Top 5 most-used slots
SELECT
  ps.slot_code,
  COUNT(b.booking_id) AS total_bookings,
  SUM(b.amount) AS revenue_generated
FROM bookings b
JOIN parking_slots ps ON b.slot_id = ps.slot_id
GROUP BY ps.slot_id
ORDER BY total_bookings DESC
LIMIT 5;
-- Mark slot as occupied on booking
UPDATE parking_slots
SET    status = 'occupied'
WHERE  slot_id = ?;  -- ? = parameterised via PreparedStatement

-- Complete a booking (check-out)
UPDATE bookings
SET
  check_out    = NOW(),
  duration_hrs = ROUND(TIMESTAMPDIFF(MINUTE, check_in, NOW()) / 60.0, 2),
  amount       = ROUND(TIMESTAMPDIFF(MINUTE, check_in, NOW()) / 60.0 * (SELECT hourly_rate
                 FROM parking_slots WHERE slot_id = bookings.slot_id), 2),
  status       = 'completed'
WHERE  booking_id = ?;

-- Free the slot after checkout
UPDATE parking_slots
SET    status = 'available'
WHERE  slot_id = (
  SELECT slot_id FROM bookings WHERE booking_id = ?
);

-- Cancel a booking
UPDATE bookings SET status = 'cancelled' WHERE booking_id = ?;

-- Delete old completed bookings (>30 days)
DELETE FROM bookings
WHERE  status = 'completed'
  AND  check_out < DATE_SUB(NOW(), INTERVAL 30 DAY);
-- Stored Procedure: Book a Slot
DELIMITER //
CREATE PROCEDURE sp_book_slot(
  IN  p_slot_id   INT,
  IN  p_user_id   INT,
  IN  p_vehicle_id INT,
  OUT p_booking_id INT,
  OUT p_msg       VARCHAR(100)
)
BEGIN
  DECLARE v_status VARCHAR(20);

  START TRANSACTION;

  SELECT status INTO v_status
  FROM  parking_slots
  WHERE slot_id = p_slot_id
  FOR UPDATE;  -- row lock

  IF v_status != 'available' THEN
    SET p_msg = 'SLOT_NOT_AVAILABLE';
    ROLLBACK;
  ELSE
    INSERT INTO bookings (slot_id, user_id, vehicle_id, check_in, status)
    VALUES (p_slot_id, p_user_id, p_vehicle_id, NOW(), 'active');

    SET p_booking_id = LAST_INSERT_ID();

    UPDATE parking_slots SET status = 'occupied'
    WHERE  slot_id = p_slot_id;

    SET p_msg = 'SUCCESS';
    COMMIT;
  END IF;
END//
DELIMITER ;

-- Call it:
CALL sp_book_slot(3, 1, 1, @booking_id, @msg);
SELECT @booking_id, @msg;
-- Trigger: auto-free slot when booking cancelled
DELIMITER //
CREATE TRIGGER trg_booking_cancel
AFTER UPDATE ON bookings
FOR EACH ROW
BEGIN
  IF NEW.status = 'cancelled' AND OLD.status = 'active' THEN
    UPDATE parking_slots
    SET    status = 'available'
    WHERE  slot_id = NEW.slot_id;
  END IF;
END//
DELIMITER ;

-- Trigger: auto-insert payment row on booking complete
DELIMITER //
CREATE TRIGGER trg_auto_payment
AFTER UPDATE ON bookings
FOR EACH ROW
BEGIN
  IF NEW.status = 'completed' AND OLD.status = 'active' THEN
    INSERT INTO payments (booking_id, amount, status)
    VALUES (NEW.booking_id, NEW.amount, 'pending');
  END IF;
END//
DELIMITER ;

import java.sql.*;

public class DBConnection {
  private static final String URL =
    "jdbc:mysql://localhost:3306/smartpark_db";
  private static final String USER = "root";
  private static final String PASS = "password";

  private static Connection conn = null;

  public static Connection getConnection()
      throws SQLException {
    if (conn == null || conn.isClosed()) {
      try {
        Class.forName(
          "com.mysql.cj.jdbc.Driver");
        conn = DriverManager.getConnection(
          URL, USER, PASS);
      } catch (ClassNotFoundException e) {
        throw new SQLException(
          "Driver not found", e);
      }
    }
    return conn;
  }

  public static void close() {
    try {
      if (conn != null) conn.close();
    } catch (SQLException e) {
      e.printStackTrace();
    }
  }
}

public class ParkingSlot {
  private int    slotId;
  private String slotCode;
  private int    floorId;
  private String slotType;
  private String status;
  private double hourlyRate;

  // Constructor
  public ParkingSlot(int id, String code,
      int floor, String type,
      String status, double rate) {
    this.slotId     = id;
    this.slotCode   = code;
    this.floorId    = floor;
    this.slotType   = type;
    this.status     = status;
    this.hourlyRate = rate;
  }

  // Getters & Setters
  public int    getSlotId()    { return slotId; }
  public String getSlotCode() { return slotCode; }
  public String getStatus()   { return status; }
  public void   setStatus(String s)
                              { status = s; }
  public double getRate()     { return hourlyRate; }

  public boolean isAvailable() {
    return "available".equals(status);
  }
}
📋 SlotDAO.java (DAO Layer)
import java.sql.*;
import java.util.*;

public class SlotDAO {

  public List<ParkingSlot> getAvailableSlots()
      throws SQLException {
    List<ParkingSlot> list = new ArrayList<>();
    String sql = "SELECT * FROM parking_slots "
               + "WHERE status = 'available'";
    try (Connection c = DBConnection.getConnection();
         PreparedStatement ps = c.prepareStatement(sql);
         ResultSet rs = ps.executeQuery()) {
      while (rs.next()) {
        list.add(new ParkingSlot(
          rs.getInt("slot_id"),
          rs.getString("slot_code"),
          rs.getInt("floor_id"),
          rs.getString("slot_type"),
          rs.getString("status"),
          rs.getDouble("hourly_rate")));
      }
    }
    return list;
  }

  public boolean updateStatus(
      int slotId, String status)
      throws SQLException {
    String sql = "UPDATE parking_slots "
               + "SET status=? WHERE slot_id=?";
    try (Connection c = DBConnection.getConnection();
         PreparedStatement ps = c.prepareStatement(sql)) {
      ps.setString(1, status);
      ps.setInt(2, slotId);
      return ps.executeUpdate() > 0;
    }
  }
}
🔖 BookingService.java (Service)
import java.sql.*;

public class BookingService {
  private final SlotDAO slotDAO = new SlotDAO();

  public int bookSlot(
      int slotId, int userId,
      int vehicleId)
      throws Exception {

    Connection conn =
      DBConnection.getConnection();
    conn.setAutoCommit(false);
    try {
      String lock =
        "SELECT status FROM parking_slots " +
        "WHERE slot_id=? FOR UPDATE";
      PreparedStatement ps =
        conn.prepareStatement(lock);
      ps.setInt(1, slotId);
      ResultSet rs = ps.executeQuery();
      if (!rs.next() ||
          !"available".equals(
            rs.getString(1))) {
        throw new Exception(
          "Slot not available");
      }
      String ins =
        "INSERT INTO bookings " +
        "(slot_id,user_id,vehicle_id," +
        "check_in,status) " +
        "VALUES(?,?,?,NOW(),'active')";
      PreparedStatement ins_ps =
        conn.prepareStatement(ins,
          Statement.RETURN_GENERATED_KEYS);
      ins_ps.setInt(1, slotId);
      ins_ps.setInt(2, userId);
      ins_ps.setInt(3, vehicleId);
      ins_ps.executeUpdate();
      int bid = ins_ps
        .getGeneratedKeys().getInt(1);

      slotDAO.updateStatus(
        slotId, "occupied");
      conn.commit();
      return bid;
    } catch (Exception e) {
      conn.rollback(); throw e;
    }
  }
}
